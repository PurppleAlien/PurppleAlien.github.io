%% ── PROYECTO CLAVE ─────────────────────────────────────────────────────
\section*{Proyecto Terminal I: Sistema de Detecci\'{o}n de Par\'{a}frasis y Plagio}
\justifying
\small
Sistema NLP de doble modo (clasificaci\'{o}n por pares + b\'{u}squeda en corpus) para detecci\'{o}n de plagio acad\'{e}mico en la UAM-I. Desplegado sobre $\sim$5{,}000 tesis con un \'{i}ndice sem\'{a}ntico FAISS de \textbf{2.28M vectores al 100\% de cobertura}.
\vspace{0.12cm}
\begin{itemize}
    \item \textbf{Arquitectura:} Bi-encoder MiniLM + FAISS $\rightarrow$ cross-encoder XLM-RoBERTa/BETO $\rightarrow$ \'{a}rbitro LLM opcional (Llama 3.3 70B) en la zona de duda $[0.35, 0.65]$.
    \item \textbf{Modelos:} XLM-RoBERTa fine-tuned y calibrado (T=5.15, ECE 0.076), BETO v8a (F1=0.889), ensamble soft-voting con \textbf{AUROC 0.999}. Evaluaci\'{o}n rigurosa con intervalos bootstrap y auditor\'{i}a de fuga de datos.
    \item \textbf{Hallazgo:} El corpus CBI es $\sim$97\% original (solo $\sim$3\% par\'{a}frasis real). El aporte central es eficiencia de datos: \textbf{60 pares humanos igualan a 180K pares algor\'{i}tmicos}.
    \item \textbf{Entrega:} App web unificada con Docker, PWA, reportes por secciones, mapas de calor y explicabilidad LLM.
\end{itemize}
\vspace{0.1cm}
{\small\itshape\color{textsec} Stack: Python, PyTorch, Hugging Face Transformers, sentence-transformers, FAISS, MongoDB, Docker.}

%% ── KEY PROJECT ─────────────────────────────────────────────────────────
\section*{Capstone Project I: Paraphrase \& Plagiarism Detection System}
\justifying
\small
End-to-end, dual-mode NLP system (pairwise classification + corpus search) for academic plagiarism detection at UAM-I. Deployed over $\sim$5{,}000 theses with a FAISS semantic index of \textbf{2.28M vectors at 100\% corpus coverage}.
\vspace{0.12cm}
\begin{itemize}
    \item \textbf{Architecture:} Bi-encoder MiniLM + FAISS $\rightarrow$ cross-encoder XLM-RoBERTa/BETO $\rightarrow$ optional LLM arbiter (Llama 3.3 70B) in the decision boundary $[0.35, 0.65]$.
    \item \textbf{Models:} Fine-tuned XLM-RoBERTa (calibrated, T=5.15, ECE 0.076), BETO v8a (F1=0.889), soft-voting ensemble with \textbf{AUROC 0.999}. Rigorous evaluation with bootstrapped confidence intervals and data-leakage audit.
    \item \textbf{Key finding:} The CBI corpus is $\sim$97\% original (only $\sim$3\% real paraphrase). Core contribution is data efficiency: \textbf{60 human pairs match 180K algorithmic pairs}.
    \item \textbf{Delivery:} Unified web app with Docker, PWA, section-level similarity reports, heatmaps, and LLM-powered explainability.
\end{itemize}
\vspace{0.1cm}
{\small\itshape\color{textsec} Stack: Python, PyTorch, Hugging Face Transformers, sentence-transformers, FAISS, MongoDB, Docker.}
